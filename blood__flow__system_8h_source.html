172// ===========================================================================
173// Main class
174//
175// PARALLEL MODEL
176// --------------
177// The mesh is a parallel::fullydistributed::Triangulation built from a
178// description that is generated on a small number of group masters, so no rank
179// ever stores the entire fine mesh. The ghost layer is the standard
180// vertex-adjacent one, which in 1-D means that a rank owning any cell incident
181// to a vertex also stores every other cell incident to that vertex. Junction
182// coupling therefore never needs data beyond the locally relevant range.
183//
184// Assembly obeys a single rule: a rank writes exactly those rows of the
185// residual and of the Jacobian whose index it owns, and reads everything else
186// from ghosted vectors. The rows of a junction with K incident vessels are
187// distributed over the incident half-faces rather than assigned to one owner:
188//
189// row u_hat_i : Riemann compatibility along vessel i, i = 0 ... K-1
190// row a_hat_0 : mass conservation over the whole junction
191// row a_hat_i : total-head continuity vessel 0 <-> i, i = 1 ... K-1
192//
193// Each of these rows belongs to the rank owning vessel i's incident cell, and
194// that rank can read all K states from its ghost layer. Every contribution is
195// consequently produced by exactly one rank and no reduction is needed to
196// close the junction equations.
197// ===========================================================================

397 // -----------------------------------------------------------------------
398 // Trace continuity
399 //
400 // Each cell carries its own (A_hat, U_hat) at every face. On an ordinary
401 // interior face the two sides are tied together: the canonical side is
402 // registered in face_dof_map and carries the Riemann equation, while the
403 // opposite side is slaved by the algebraic rows
404 // A_hat_dup - A_hat_canon = 0 , U_hat_dup - U_hat_canon = 0 .
405 // Junction and boundary half-faces are each their own canonical owner and
406 // are untouched by this mechanism.
407 //
408 // A pair is stored on the rank owning the duplicate side, which is the rank
409 // that owns the two rows; the canonical indices are read from the ghosted
410 // solution.
411 // -----------------------------------------------------------------------

413 {
414 types::global_dof_index a_dup, u_dup; // slaved (duplicate) side
415 types::global_dof_index a_canon, u_canon; // master (canonical) side
416 };

419 // Return the two global trace DoFs (component 2 -> A_hat, component 3 ->
420 // U_hat) with support on local face `f`, given a cell's local dof indices.
421 // In 1-D, FE_DGQ(1) has exactly one such DoF per component per face.
422 std::pair<types::global_dof_index, types::global_dof_index>
423 face_trace_dofs(const std::vector<types::global_dof_index> &ldofs,
424 const unsigned int f) const
425 {
426 types::global_dof_index a = numbers::invalid_dof_index;
427 types::global_dof_index u = numbers::invalid_dof_index;
428 for (unsigned int i = 0; i < fe->n_dofs_per_cell(); ++i)
429 {
430 const unsigned int c = fe->system_to_component_index(i).first;
431 if ((c == 2 || c == 3) && fe->has_support_on_face(i, f))
432 (c == 2 ? a : u) = ldofs[i];
433 }
434 return {a, u};
435 }

509 // Do the two vessels share an identical pressure law?
510 bool
511 two_way_pressure_laws_match(const unsigned int vid_a,
512 const unsigned int vid_b) const
513 {
514 if (!vessel_map.count(vid_a) || !vessel_map.count(vid_b))
515 return false;
516 const auto &A = vessel_map.at(vid_a);
517 const auto &B = vessel_map.at(vid_b);
518 const auto eq = [](const double x, const double y) {
519 return std::abs(x - y) <=
520 1e-10 * std::max(1.0, std::max(std::abs(x), std::abs(y)));
521 };
522 return eq(A.a0, B.a0) && eq(A.a_d, B.a_d) && eq(A.E, B.E) &&
523 eq(A.h_wall, B.h_wall) && eq(A.p0, B.p0) && eq(A.p_d, B.p_d) &&
524 eq(A.r_in, B.r_in) && eq(A.r_out, B.r_out);
525 }

664 // Internal helpers — physics / constitutive law
665 // -----------------------------------------------------------------------
666
667 // Local diastolic cross-sectional area at a cell's centroid, obtained by
668 // interpolating linearly in arc length between the inlet radius r_in and the
669 // outlet radius r_out of the parent vessel. The arc-length bounds are the
670 // global ones, so the value a cell gets does not depend on the partitioning.
671 //
672 // If r_in == r_out == 0 (no taper data), fall back to the vessel-level a_d.
673 double
674 compute_a_d_local(
675 const typename DoFHandler<dim, spacedim>::active_cell_iterator &cell) const
676 {
677 const unsigned int vid = cell->material_id();
678 const auto &vpp = vessel_map.at(vid);
679
680 if (vpp.r_in <= 0.0 || vpp.r_out <= 0.0)
681 return vpp.a_d;
682
683 const auto &[s_min, s_max] = vessel_s_bounds.at(vid);
684
685 const Tensor<1, spacedim> d_hat = compute_directional_vector(cell);
686
687 const double s = cell->center() * d_hat;
688
689 const double xi_geometry =
690 (s_max - s_min > 1e-14) ? (s - s_min) / (s_max - s_min) : 0.5;
691
692 const double r = vpp.r_in + xi_geometry * (vpp.r_out - vpp.r_in);
693
694 return numbers::PI * r * r;
695 }

703 const auto &vpp = vessel_map.at(vid);
704 if (vpp.r_in <= 0.0 || vpp.r_out <= 0.0)
705 return vpp.a_d;
706
707 const auto &[s_min, s_max] = vessel_s_bounds.at(vid);
708 const Tensor<1, spacedim> d_hat = compute_directional_vector(cell);
709 const double s = cell->vertex(face_no) * d_hat;

712 const double r = vpp.r_in + xi_geometry * (vpp.r_out - vpp.r_in);
713 return numbers::PI * r * r;
714 }
