cmake_minimum_required(VERSION 3.23)
project(MetricFlowXConsumer LANGUAGES CXX)
find_package(MetricFlowX CONFIG REQUIRED)
# Repeated discovery must not redefine imported targets.
find_package(MetricFlowX CONFIG REQUIRED)
deal_ii_initialize_cached_variables()
get_target_property(_configs MetricFlowX::metric_flow_x IMPORTED_CONFIGURATIONS)
foreach(_variant IN ITEMS DEBUG RELEASE)
  if(NOT _variant IN_LIST _configs)
    message(FATAL_ERROR "Missing installed configuration: ${_variant}")
  endif()
endforeach()
string(TOUPPER "${CMAKE_BUILD_TYPE}" _config)
get_target_property(_location MetricFlowX::metric_flow_x IMPORTED_LOCATION_${_config})
if(_config STREQUAL "DEBUG" AND NOT _location MATCHES "metric_flow_x_debug")
  message(FATAL_ERROR "Debug selected the wrong library: ${_location}")
elseif(_config STREQUAL "RELEASE" AND _location MATCHES "metric_flow_x_debug")
  message(FATAL_ERROR "Release selected the Debug library: ${_location}")
endif()
add_executable(package_smoke main.cc)
deal_ii_setup_target(package_smoke)
target_link_libraries(package_smoke MetricFlowX::metric_flow_x)
enable_testing()
add_test(NAME package_smoke COMMAND package_smoke)
