cmake_minimum_required(VERSION 3.23)
project(HepaticImportSmoke LANGUAGES CXX)
find_package(deal.II 9.5 REQUIRED HINTS ${DEAL_II_DIR})
deal_ii_initialize_cached_variables()
if(NOT DEAL_II_WITH_VTK)
  message(FATAL_ERROR "This check requires deal.II with VTK")
endif()
get_filename_component(METRIC_FLOW_X_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(hepatic_import_smoke main.cc "${METRIC_FLOW_X_ROOT}/source/io/vtk_utils.cc")
target_include_directories(hepatic_import_smoke PRIVATE "${METRIC_FLOW_X_ROOT}/include")
deal_ii_setup_target(hepatic_import_smoke)
